        "ctrl+k", "toggle_vim",   // 切换 Vim 模式
        "ctrl+v", "toggle_voice", // 切换语音输入
        "ctrl+t", "tasks"         // 显示后台任务

/**
 * 对标 Python resolve_keybindings()。
 * 用户自定义覆盖默认值。
 */
public class KeybindingsResolver {

    public Map<String, String> resolve(Path userConfigPath) {
        Map<String, String> merged = new HashMap<>(DefaultKeybindings.DEFAULTS);

        // 加载用户自定义快捷键（~/.openharness/keybindings.json）
        if (Files.exists(userConfigPath)) {
            Map<String, String> userBindings = objectMapper.readValue(
                userConfigPath.toFile(),
                new TypeReference<Map<String, String>>() {}
            );
            merged.putAll(userBindings); // 用户覆盖默认
        }

        // 验证：所有 action 必须已知
        Set<String> knownActions = Set.of("clear", "toggle_vim", "toggle_voice",
            "tasks", "help", "compact", "exit", "model");
        for (var entry : merged.entrySet()) {
            if (!knownActions.contains(entry.getValue())) {
                logger.warn("Unknown keybinding action: {} -> {}", entry.getKey(), entry.getValue());
            }
        }

        return Map.copyOf(merged);
    }
}

## 4.27 Token 估算方案

Java 生态没有 Python tiktoken 库的直接等价物。Python 版 OpenHarness 本身使用简单启发式：`max(1, (len(text) + 3) // 4)` 即字符数 / 4。

/**
 * Token 计数——Java 实现方案。
 * 对标 Python estimate_tokens()。
 */
public class TokenEstimator {

    /**
     * 方案 A（默认，与 Python 对齐）：字符 / 4 启发式。
     * 简单、零依赖，Python 版使用此方案。
     */
    public static long estimateTokens(String text) {
        if (text == null || text.isEmpty()) return 0;

 * PID 文件锁（防止并发写入冲突）。
 */
public class FileLock implements AutoCloseable {
    private final FileChannel channel;
    private final java.nio.channels.FileLock lock;

    public static FileLock acquire(Path lockFile) throws IOException {
        FileChannel channel = FileChannel.open(lockFile,
            StandardOpenOption.CREATE, StandardOpenOption.WRITE);
        java.nio.channels.FileLock lock = channel.tryLock();
        if (lock == null) throw new LockAcquisitionException(lockFile);
        channel.write(ByteBuffer.wrap(
            (ProcessHandle.current().pid() + "\n").getBytes()));
        return new FileLock(channel, lock, lockFile);
    }

    @Override public void close() throws IOException {
        lock.release(); channel.close(); Files.deleteIfExists(lockFile);
    }
}

/**
 * 对标 Python list_document_symbols() / workspace_symbol_search() / go_to_definition()。
 * 纯 AST 代码智能（支持 Python），后期扩展 LSP 客户端。
 */
public class CodeIntelligence {
    private final Path workspaceRoot;

    public List<SymbolLocation> searchSymbols(String query) {
        return iterSourceFiles(workspaceRoot).stream()
            .flatMap(f -> extractSymbols(readFile(f), f).stream())
            .filter(s -> s.name().toLowerCase().contains(query.toLowerCase())).toList();
    }

    public Optional<SymbolLocation> goToDefinition(String name) {
        return iterSourceFiles(workspaceRoot).stream()
            .flatMap(f -> extractSymbols(readFile(f), f).stream())
            .filter(s -> s.name().equals(name)).findFirst();
    }

    public List<SymbolLocation> findReferences(String name) {
        Pattern p = Pattern.compile("\\b" + Pattern.quote(name) + "\\b");
        return iterSourceFiles(workspaceRoot).stream()
            .flatMap(f -> grepFile(f, p).stream()).toList();
    }

    private List<SymbolLocation> extractSymbols(String source, Path file) { /* AST 遍历 */ }
    private List<Path> iterSourceFiles(Path root) {
        try (var s = Files.walk(root)) {
            return s.filter(f -> f.toString().matches(".*\\.(py|java|ts|tsx|js|go|rs)$"))
                    .filter(f -> !f.toString().contains("/venv/"))
                    .filter(f -> !f.toString().contains("/node_modules/")).toList();
        }
    }
}

public record SymbolLocation(String name, String kind, Path path,
                             int line, int character, String signature, String docstring) {}

public class MemoryExtractionService {
    /**
     * 安全守卫：如果当前轮已写入过文件则跳过提取。
     */
    public boolean hasMemoryWritesSince(List<ConversationMessage> recentMessages) {
        Set<String> writeTools = Set.of("write_file", "edit_file");
        return recentMessages.stream().flatMap(m -> m.content().stream())
            .filter(b -> b instanceof ToolUseBlock)
            .anyMatch(b -> writeTools.contains(((ToolUseBlock) b).name()));
    }

    /**
     * 从对话轮次提取持久记忆。
     * 1. 安全守卫检查 2. 构建提取 prompt 3. 调用 LLM 4. 解析 JSON 记录
     */
    public List<ExtractionRecord> extract(List<ConversationMessage> messages,
                                           String manifest, StreamingApiClient client) {
        if (hasMemoryWritesSince(messages)) return List.of();
        String response = client.streamMessages("claude-sonnet-4-6",
            EXTRACTION_SYSTEM_PROMPT, messages, List.of(), defaults()).collect();
        return parseExtractionRecords(response);
    }

    private static final String EXTRACTION_SYSTEM_PROMPT = """
        Extract durable facts as JSON: [{"name":"...","type":"user|feedback|project|reference","content":"..."}]
        Skip transient/in-progress items. Avoid duplicates with existing memories below.
        """;
}

public record ExtractionRecord(String name, String type, String content) {}
