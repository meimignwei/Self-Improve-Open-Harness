 * 对标 Python BaseTool。
 * 所有44个内置工具的基类。

public abstract class BaseTool<T> {

    public abstract String getName();
    public abstract String getDescription();
    public abstract Class<T> getInputModel();
    public abstract boolean requiresApproval();
    public abstract boolean isReadOnly(T arguments);

    /**
     * 执行工具逻辑。
     * @return ToolResult 包含文本输出、是否错误、可选媒体
     */
    public abstract ToolResult execute(T arguments, ToolContext context);

    /**
     * 生成 JSON Schema（用于 LLM function calling）。
     */
    public final JsonNode getInputSchema() {
        return JsonSchemaGenerator.generate(getInputModel());
    }
}

public record ToolResult(
        String content,
        boolean isError,
        List<String> mediaFiles
) {}

public record ToolContext(
        String sessionId,
        Path workingDirectory,
        Settings settings,
        PermissionChecker permissions,
        RuntimeBundle runtime
) {}

### 4.3.3 循环依赖解耦示例：AgentTool

```java
/**
 * 对标 Python AgentTool。
 * 关键设计：只依赖 openharness-common 中的 AgentRuntime 接口，
 * 而非直接依赖 openharness-engine 中的 QueryEngine 实现类。
 * 这打破了 engine ↔ tools 模块间的循环依赖。
 */
public class AgentTool extends BaseTool<AgentInput> {

    private final AgentRuntime agentRuntime;

    public AgentTool(AgentRuntime agentRuntime) {
        this.agentRuntime = agentRuntime;
    }

    @Override
    public ToolResult execute(AgentInput input, ToolContext context) {
        List<ConversationMessage> messages = buildSubAgentMessages(input);
        QueryOptions options = QueryOptions.builder()
                .model(input.model())
                .systemPromptOverride(input.system())
                .build();

        // 通过接口调用，无需知道 QueryEngine 的存在
        List<StreamEvent> events = PublisherAdapter.toList(
                agentRuntime.runQuery(messages, options));

        String response = events.stream()
                .filter(e -> e instanceof AssistantTextDelta)
                .map(e -> ((AssistantTextDelta) e).text())
                .collect(Collectors.joining());

        return new ToolResult(response, false, List.of());
    }
}
```

### 4.3.4 关键工具实现示例：BashTool

```java
/**
 * 对标 Python BashTool。
 * 支持交互命令预检、输出截断、沙箱路由。
 */
public class BashTool extends BaseTool<BashInput> {

    private static final int MAX_OUTPUT_LENGTH = 12_000;
    private static final Set<String> INTERACTIVE_COMMANDS = Set.of(
            "create-next-app", "npm init", "npx create-", "yarn create"
    );

    @Override
    public ToolResult execute(BashInput input, ToolContext context) {
        Span span = tracer.spanBuilder("tool.bash.execute")
                .setAttribute("tool.command", truncate(input.command(), 200))
                .setAttribute("tool.timeout_ms", input.timeoutMs())
                .startSpan();

        try (var scope = span.makeCurrent()) {
            // 1．交互命令预检
            if (isInteractiveCommand(input.command())) {
                return ToolResult.error("Interactive command detected. Use --yes or equivalent flags.");
            }

            // 2．沙箱路由
            String[] cmd = context.settings().sandbox().isEnabled()
                    ? SandboxAdapter.wrapCommand(input.command(), context)
                    : new String[]{"/bin/bash", "-c", input.command()};

            // 3．进程执行
            ProcessBuilder pb = new ProcessBuilder(cmd)
                    .directory(context.workingDirectory().toFile())
                    .redirectErrorStream(true);
            Process process = pb.start();

            String output = readWithTimeout(process, input.timeoutMs());

            // 4．输出截断
            if (output.length() > MAX_OUTPUT_LENGTH) {
                output = output.substring(0, MAX_OUTPUT_LENGTH) + "\n... (truncated)";
            }

            int exitCode = process.exitValue();

            // 指标
            Counter.builder("tool.bash.executions")
                    .tag("exit_code", String.valueOf(exitCode))
                    .register(meters)
                    .increment();
            span.setAttribute("tool.exit_code", exitCode);
            return new ToolResult(output, exitCode != 0, List.of());

        } catch (Exception e) {
            span.recordException(e);
            span.setStatus(StatusCode.ERROR);
            return ToolResult.error(e.getMessage());
        } finally {
            span.end();
        }
    }
}
```

## 4.4 权限系统（openharness-permissions）

**对标**：`src/openharness/permissions/`(3个文件）

```java
public class PermissionChecker {

    /**
     * 检查工具调用权限。
     * 三种模式：DEFAULT（写操作需审批）、PLAN（编辑自动通过、bash 需审批）、FULL AUTO（全部通过）
     */
    public PermissionResult checkToolPermission(
            String toolName,
            JsonNode arguments,
            PermissionMode mode,
            List<PathRule> pathRules
    ) {
        Span span = tracer.spanBuilder("permission.check")
                .setAttribute("tool.name", toolName)
                .setAttribute("permission.mode", mode.name())
                .startSpan();

        try {
            // 1.FULL_AUTO 模式直接放行
            if (mode == PermissionMode.FULL_AUTO) {
                return PermissionResult.ALLOWED;
            }

            // 2．敏感路径检查（~/.ssh,~/.aws,~/.gnupg等）
            Optional<String> targetPath = extractTargetPath(toolName, arguments);
            if (targetPath.isPresent() && isSensitivePath(targetPath.get())) {
                Counter.builder("permission.sensitive_path.blocked")
                        .register(meters)
                        .increment();
                return PermissionResult.denied("Access to sensitive path is blocked");
            }

            // 3．路径级规则匹配
            if (targetPath.isPresent()) {
                for (PathRule rule : pathRules) {
                    if (rule.matches(targetPath.get())) {
                        return rule.isAllow() ?
                                PermissionResult.ALLOWED : PermissionResult.denied(rule.reason());
                    }
                }
            }

            // 4．模式级检查
            BaseTool<?> tool = toolRegistry.get(toolName);
            if (tool.isReadOnly(arguments)) {
                return PermissionResult.ALLOWED;
            }
            if (mode == PermissionMode.PLAN && isFileEditTool(toolName)) {
                return PermissionResult.ALLOWED;
            }

            return PermissionResult.NEEDS_APPROVAL;

        } finally {
            span.end();
        }
    }
}
```
