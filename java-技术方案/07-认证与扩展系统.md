### 4.12.2 Hook 执行器

```java
/**
 * 对标 Python HookExecutor。
 * 主执行引擎：根据 Hook 类型分发到对应的 handler。
 */
public class HookExecutor {
    private final HookRegistry registry;
    private final HookExecutionContext ctx; // cwd, apiClient, defaultModel

    public AggregatedHookResult execute(HookEvent event, Map<String, Object> payload) {
        List<HookDefinition> hooks = registry.get(event);
        List<HookResult> results = new ArrayList<>();

        // 按优先级排序执行
        for (HookDefinition hook : hooks.stream().sorted(byPriority()).toList()) {
            if (!matchesHook(hook, event, payload)) continue;

            HookResult result = switch (hook) {
                case CommandHook ch -> runCommandHook(ch, event, payload);
                case HttpHook hh -> runHttpHook(hh, event, payload);
                case PromptHook ph -> runPromptLikeHook(ph, event, payload);
                case AgentHook ah -> runPromptLikeHook(ah, event, payload);
            };
            results.add(result);

            // blockOnFailure: 终止后续 Hook 执行
            if (!result.success() && hook.blockOnFailure()) break;
        }
        return new AggregatedHookResult(results);
    }

    // 命令型：通过 ProcessBuilder 启动子进程
    // 注入环境变量 OPENHARNESS_HOOK_EVENT / OPENHARNESS_HOOK_PAYLOAD
    private HookResult runCommandHook(CommandHook hook, HookEvent event, Map<String,Object> payload) { ... }

    // HTTP 型：POST JSON payload 到指定 URL
    private HookResult runHttpHook(HttpHook hook, HookEvent event, Map<String,Object> payload) { ... }

    // Prompt/Agent 型：调用 LLM 验证条件，期望返回 {"ok": true/false}
    private HookResult runPromptLikeHook(HookDefinition hook, HookEvent event, Map<String,Object> payload) { ... }
}

public record AggregatedHookResult(List<HookResult> results) {
    public boolean blocked() { return results.stream().anyMatch(r -> r.blocked()); }
    public String reason() { /* 聚合所有 blocked 原因 */ }
}
```

```java
    private final Map<HookEvent, List<HookDefinition>> hooks = new HashMap<>();

    public void register(HookEvent event, HookDefinition hook) {
        hooks.computeIfAbsent(event, k -> new ArrayList<>()).add(hook);
```

```java
/**
 * 对标 Python load_skill_registry()。
 * 按优先级加载：bundled → user → extra_dirs → project → plugin
 */
public class SkillLoader {

    public SkillRegistry loadAll(Settings settings, List<LoadedPlugin> plugins) {
        SkillRegistry registry = new SkillRegistry();

        // 1. 内置技能（jar 内 resources/skills/）
        loadFromResourceDir(registry, "skills/", "bundled");

        // 2. 用户技能（~/.openharness/skills/）
        loadFromDir(registry, Paths.getHomeSkillsDir(), "user");

        // 3. 额外技能目录（settings.extraSkillDirs）
        for (String dir : settings.extraSkillDirs()) {
            loadFromDir(registry, Path.of(dir), "user");
        }

        // 4. 项目技能（cwd → git root 的 .openharness/skills/）
        Path projectRoot = findProjectRoot(cwd); // 向上查找 .git 目录
        if (projectRoot != null) {
            loadFromDir(registry, projectRoot.resolve(".openharness/skills"), "project");
            loadFromDir(registry, projectRoot.resolve(".claude/skills"), "project");
            loadFromDir(registry, projectRoot.resolve(".agents/skills"), "project");
        }

        // 5. 插件技能
        for (LoadedPlugin plugin : plugins) {
            if (plugin.skillsDir() != null) {
                loadFromDir(registry, plugin.skillsDir(), "plugin:" + plugin.name());
            }
        }

        return registry;
    }

    // 遍历目录，查找 <dir>/SKILL.md 文件
    private void loadFromDir(SkillRegistry registry, Path dir, String source) {
        if (!Files.isDirectory(dir)) return;
        try (var files = Files.walk(dir, 3)) {
            files.filter(f -> f.getFileName().toString().equals("SKILL.md"))
                 .forEach(skillMd -> {
                     SkillDefinition skill = parseSkillMd(skillMd, source);
                     registry.register(skill);
                 });
        }
    }
}
```

```java
/**
 * 对标 Python load_plugins()。
 * 从用户目录和项目目录发现并加载插件。
 */
public class PluginLoader {

    public List<LoadedPlugin> loadAll(Settings settings) {
        List<LoadedPlugin> plugins = new ArrayList<>();

        // 1. 发现路径
        List<Path> discoverDirs = List.of(
            Paths.getUserHome().resolve(".openharness/plugins"),   // 用户级
            findProjectRoot(cwd).resolve(".openharness/plugins")   // 项目级
        );

        for (Path discoverDir : discoverDirs) {
            if (!Files.isDirectory(discoverDir)) continue;
            try (var dirs = Files.list(discoverDir)) {
                dirs.filter(Files::isDirectory).forEach(pluginDir -> {
                    PluginManifest manifest = loadManifest(pluginDir);
                    boolean enabled = settings.enabledPlugins().contains(manifest.name())
                                   || manifest.enabledByDefault();
                    if (enabled) {
                        plugins.add(loadPluginArtifacts(pluginDir, manifest));
                    }
                });
            }
        }
        return plugins;
    }

    // 加载插件的所有工件
    private LoadedPlugin loadPluginArtifacts(Path dir, PluginManifest manifest) {
        return new LoadedPlugin(manifest, dir, true,
            loadSkills(dir.resolve(manifest.skillsDir())),
            loadCommands(dir.resolve(manifest.commandsDir())),
            loadAgents(dir.resolve(manifest.agentsDir())),
            loadTools(dir.resolve(manifest.toolsDir())),       // 动态加载 .class / .py
            loadHooks(dir.resolve(manifest.hooksFile())),
            loadMcpConfig(dir.resolve(manifest.mcpFile()))
        );
    }
}

/**
```
```java
 * 对标 Python CommandRegistry。
 * 管理所有 /slash 命令的注册、别名解析和分发。
 */
public class CommandRegistry {
    private final Map<String, SlashCommand> commands = new LinkedHashMap<>();
    private final Map<String, String> aliases = new HashMap<>();

    public void register(SlashCommand command) {
        commands.put(command.name(), command);
        for (String alias : command.aliases()) {
            aliases.put(alias, command.name());
        }
    }

    /**
     * 解析用户输入："/model claude-sonnet-4-6" → command="model", args="claude-sonnet-4-6"
     */
    public Optional<SlashCommand> lookup(String rawInput) {
        if (!rawInput.startsWith("/")) return Optional.empty();
        String[] parts = rawInput.substring(1).split("\\s+", 2);
        String cmdName = parts[0];
        String cmdKey = aliases.getOrDefault(cmdName, cmdName);
        return Optional.ofNullable(commands.get(cmdKey));
    }

    public String helpText() { /* 生成所有命令的帮助文本 */ }
    public List<SlashCommand> listCommands() { return List.copyOf(commands.values()); }
}

/**
```
```java
 * 对标 Python McpClientManager。
 * 管理所有 MCP 服务器的连接、工具调用和资源读取。
 */
public class McpClientManager {
    private final Map<String, McpConnectionState> connections = new ConcurrentHashMap<>();

    /**
     * 连接所有配置的 MCP 服务器（settings + plugins）。
     */
    public void connectAll(List<McpServerConfig> configs) {
        for (McpServerConfig config : configs) {
            try {
                McpConnectionState state = switch (config) {
                    case McpStdioServerConfig sc -> connectStdio(sc);
                    case McpHttpServerConfig hc -> connectHttp(hc);
                    case McpWebSocketServerConfig wc -> connectWs(wc);
                };
                connections.put(state.name(), state);
            } catch (Exception e) {
                connections.put(config.name(), McpConnectionState.failed(config.name(), e));
            }
        }
    }

    /**
```

### 4.17.2 CLAUDE.md 发现

```java
/**
 * 对标 Python discover_claude_md_files()。
 * 从 cwd 向上遍历至 git root，收集所有 CLAUDE.md 文件。
 */
public class ClaudeMdLoader {

    public static String load(Path cwd) {
        StringBuilder sb = new StringBuilder();
        // 1. 从 cwd 向上查找，收集路径上所有 CLAUDE.md
        // 2. 同时查找 .claude/CLAUDE.md 和 .claude/rules/*.md
        // 3. 按距离排序（最近的优先），格式化为 Markdown
        List<Path> files = discover(cwd);
        for (Path f : files) {
            sb.append("--- CLAUDE.md from ").append(f.getParent()).append(" ---\n");
            sb.append(Files.readString(f)).append("\n");
        }
        return sb.toString();
    }

    private static List<Path> discover(Path start) {
        List<Path> found = new ArrayList<>();
        Path current = start.toAbsolutePath();
        while (current != null) {
            Path claudeMd = current.resolve("CLAUDE.md");
            if (Files.exists(claudeMd)) found.add(claudeMd);
            // .claude/CLAUDE.md, .claude/rules/*.md
            Path claudeDir = current.resolve(".claude/CLAUDE.md");
            if (Files.exists(claudeDir)) found.add(claudeDir);
            // 停止条件：git root
            if (Files.exists(current.resolve(".git"))) break;
            current = current.getParent();
        }
        return found;
    }
}
```
